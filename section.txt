function flat(v){return new THREE.Vector3(v.x,GROUND_Y,v.z)}
function heading(from,to){return Math.atan2(to.x-from.x,to.z-from.z)}
function normalizeAngle(a){while(a>Math.PI)a-=Math.PI*2;while(a<-Math.PI)a+=Math.PI*2;return a}
function rotateTo(obj,to,turnRate=4.8){return new Promise(resolve=>{
  const target=heading(obj.position,to), start=obj.rotation.y, delta=normalizeAngle(target-start);
  const duration=Math.max(70,Math.abs(delta)/turnRate*1000), t0=performance.now();
  function frame(now){const t=Math.min(1,(now-t0)/duration);obj.rotation.set(0,start+delta*t,0);obj.position.y=GROUND_Y;if(t<1)requestAnimationFrame(frame);else{obj.rotation.set(0,target,0);resolve()}}
  requestAnimationFrame(frame)
})}
async function driveSegment(obj,to,speed=12,onFrame=null,reverse=false){
  const b=flat(to); if(flat(obj.position).distanceTo(b)<.03)return;
  const faceTarget=reverse?new THREE.Vector3(obj.position.x-(b.x-obj.position.x),0,obj.position.z-(b.z-obj.position.z)):b;
  await rotateTo(obj,faceTarget,5.5);
  return new Promise(resolve=>{
    let last=performance.now();
    function frame(now){
      const dt=Math.min(.04,(now-last)/1000);last=now;
      const dx=b.x-obj.position.x,dz=b.z-obj.position.z,dist=Math.hypot(dx,dz);
      if(dist<=.04){obj.position.set(b.x,GROUND_Y,b.z);if(onFrame)onFrame();resolve();return}
      const step=Math.min(dist,speed*dt), inv=dist?1/dist:0;
      obj.position.x+=dx*inv*step;obj.position.z+=dz*inv*step;obj.position.y=GROUND_Y;
      if(onFrame)onFrame();requestAnimationFrame(frame)
    }
    requestAnimationFrame(frame)
  })
}
